#   pip install -e ".[env]"        # gymnasium
#   pip install -e ".[memory]"     # chromadb

#   pyaif-toolkit / soul-agent     —— PyPI 上找不到对应可用包，clone 后必然 install 失败
#   chromadb==1.5.9                —— 版本号不存在
#   torch（作为必需项）             —— Python 3.13 上无对应 wheel，会直接装挂
